sympy · functions and limits
from sympy import *


── functions ──

f.subs(x, value)
(x**2 + 1).subs(x, 3) → 10      # f(value): a function is just an expression in x

f.subs(x, g)
(x**2 + 1).subs(x, 2*x - 3) → (2*x - 3)**2 + 1      # f(g(x))

Lambda(x, expr)
Lambda(x, x**2)(3) → 9      # a SymPy function you can call with brackets

lambdify(x, expr)
lambdify(x, x**2 + 1)(3) → 10      # a normal Python function, for fast decimals

Piecewise((expr1, condition), (expr2, True))
Piecewise((x + 1, x < 1), (3 - x, True)).subs(x, 0) → 1      # a split function

floor(expr)   Abs(expr)
floor(Rational(17, 10)) → 1      # rounds down / the size without the minus sign


── limits ──

limit(expr, x, a)
limit(sin(x)/x, x, 0) → 1      # the limit as x → a (from the RIGHT unless you say)

limit(expr, x, a, "-")
limit(1/x, x, 0, "-") → -oo      # from the left

limit(expr, x, a, "+-")
limit(x**2, x, 3, "+-") → 9      # from both sides (zoo means the sides disagree)

limit(expr, x, oo)
limit((3*x**2 + 1)/(x**2 - 5), x, oo) → 3      # as x → infinity

Limit(expr, x, a).doit()
Limit(sin(x)/x, x, 0).doit() → 1      # writes the limit down, then .doit() works it out

nan   zoo
((x**2 - 4)/(x - 2)).subs(x, 2) → nan      # "not a number" (0/0) / complex infinity
