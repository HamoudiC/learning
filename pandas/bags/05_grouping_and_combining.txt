pandas · grouping and combining
import pandas as pd


── one answer per group ──

df.groupby("col")["other"].sum()
pd.DataFrame({"g": ["a", "a", "b"], "v": [1, 2, 5]}).groupby("g")["v"].sum().to_dict() → {'a': 3, 'b': 5}      # a total for each group

df.groupby(a_series)["other"].sum()
pd.DataFrame({"v": [1, 2, 5]}).groupby(pd.Series(["a", "a", "b"]))["v"].sum().to_dict() → {'a': 3, 'b': 5}      # groups by values you worked out, row by row

.mean()   .count()   .max()   .min()
pd.DataFrame({"g": ["a", "a"], "v": [1, 3]}).groupby("g")["v"].mean().to_dict() → {'a': 2.0}      # other answers per group (count skips missing)

df.groupby("col").size()
pd.DataFrame({"g": ["a", "a", "b"]}).groupby("g").size().to_dict() → {'a': 2, 'b': 1}      # how many rows in each group

df.groupby("col").agg(new=("col", "how"), ...)
pd.DataFrame({"g": ["a", "a"], "v": [1, 3]}).groupby("g").agg(total=("v", "sum")).to_dict("list") → {'total': [4]}      # several answers, with your own column names

result.reset_index()
pd.DataFrame({"g": ["a"], "v": [1]}).groupby("g")["v"].sum().reset_index().columns.tolist() → ['g', 'v']      # turns the group labels back into a column

result.idxmax()
pd.Series([3, 9], index=["a", "b"]).idxmax() → 'b'      # the group with the biggest answer


── counting ──

s.value_counts()
pd.Series(["a", "b", "a"]).value_counts().to_dict() → {'a': 2, 'b': 1}      # how many times each value appears, most first

s.nunique()
pd.Series(["a", "b", "a"]).nunique() → 2      # how many different values

s.unique()
pd.Series(["b", "a", "b"]).unique().tolist() → ['b', 'a']      # each different value once, in the order they first appear

pd.crosstab(df["a"], df["b"])
pd.crosstab(pd.Series(["x", "x"]), pd.Series(["p", "q"])).loc["x"].to_dict() → {'p': 1, 'q': 1}      # a count table: rows from a, columns from b


── joining tables ──

pd.merge(left, right, on="key")
len(pd.merge(pd.DataFrame({"k": [1, 2]}), pd.DataFrame({"k": [2, 3]}), on="k")) → 1      # only rows whose key is in BOTH

pd.merge(left, right, on="key", how="left")
len(pd.merge(pd.DataFrame({"k": [1, 2]}), pd.DataFrame({"k": [2, 3]}), on="k", how="left")) → 2      # every left row (no match → NaN)

pd.merge(left, right, on="key", how="outer")
len(pd.merge(pd.DataFrame({"k": [1, 2]}), pd.DataFrame({"k": [2, 3]}), on="k", how="outer")) → 3      # every row of both

pd.merge(..., how="outer", indicator=True)
pd.merge(pd.DataFrame({"k": [1]}), pd.DataFrame({"k": [2]}), on="k", how="outer", indicator=True)["_merge"].tolist() → ['left_only', 'right_only']      # adds a _merge column saying where each row came from

pd.concat([df1, df2], ignore_index=True)
pd.concat([pd.DataFrame({"a": [1]}), pd.DataFrame({"a": [2]})], ignore_index=True)["a"].tolist() → [1, 2]      # one table under the other, numbered 0, 1, 2
